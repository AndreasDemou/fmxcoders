############################################
# ICML Polynomial SAE – Training + SAEBench Commands
# All experiments use context_size=128 (Neuronpedia-style)
############################################

# One-time environment setup
export WANDB_PROJECT="icml_polynomial_sae"
export TOKENIZERS_PARALLELISM=false

############################################
# GPT-2 Small (124M) – Layer 8 – width 16k
############################################

# Vanilla TopK
python train_and_saebench.py --exp gpt2_l8 --architecture topk \
  --width 16k --k 64 --training_tokens 300000000 --context_size 128 \
  --no_rescale_by_decoder_norm \
  --checkpoint_path checkpoints/gpt2_l8_topk_16k \
  --run_saebench --use_saelens_defaults

# PolySAE TopK
python train_and_saebench.py --exp gpt2_l8 --architecture topk \
  --width 16k --k 64 --training_tokens 300000000 --context_size 128 \
  --poly --no_rescale_by_decoder_norm \
  --checkpoint_path checkpoints/gpt2_l8_topk_poly_16k \
  --run_saebench --use_saelens_defaults
